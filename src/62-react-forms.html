제어 컴포넌트

function NameInput() {
  const [name, setName] = useState("");

  return (
    <input
      value={name}                                // 값은 상태에서
      onChange={(e) => setName(e.target.value)}   // 변경은 상태로
    />
  );
}

입력 요소의 값이 React 상태에 의해 결정되는 방식입니다. 57장의 UI = f(state)가 입력 요소에도 적용된 것입니다.

🔒 value만 주고 onChange를 빠뜨리면

입력이 아무것도 안 됩니다. 타이핑해도 값이 변하지 않습니다. 상태가 안 바뀌니 화면도 그대로이기 때문입니다. React가 콘솔에 경고를 띄워 줍니다.
읽기 전용이 의도라면 readOnly를 명시하세요.

⚖️ 무엇을 쓸까

  • 입력하는 동안 무언가 해야 한다 (검색 자동완성, 실시간 검증, 글자 수 표시) → 제어
  • 제출할 때만 값이 필요하다 → 비제어가 더 간단합니다
  • 파일 업로드 → 비제어(<input type="file">은 보안상 값을 설정할 수 없습니다)

React 공식 문서는 제어 컴포넌트를 기본으로 안내합니다. 다만 입력이 많은 큰 폼에서는 비제어나 폼 라이브러리가 성능·코드량 면에서 유리합니다.

여러 입력 관리하기

하나의 객체 상태로 
function SignupForm() {
  const [form, setForm] = useState({
    name: "", email: "", password: "", agree: false
  });

  function handleChange(e) {
    const { name, value, type, checked } = e.target;
    setForm(prev => ({
      ...prev,                                   // 불변성 (59장)
      [name]: type === "checkbox" ? checked : value    // 계산된 키 (28장)
    }));
  }

  return (
    <form>
      <input name="name" value={form.name} onChange={handleChange} />
      <input name="email" type="email" value={form.email} onChange={handleChange} />
      <input name="password" type="password" value={form.password} onChange={handleChange} />
      <input name="agree" type="checkbox" checked={form.agree} onChange={handleChange} />
    </form>
  );
}

🔑 name 속성이 핵심입니다

9장에서 배운 name이 여기서 다시 등장합니다. name을 상태 객체의 키와 일치시키면 핸들러 하나로 모든 입력을 처리할 수 있습니다. 입력이 10개여도 핸들러는 하나입니다.

입력 종류읽는 값
text, email, password, textarea, selecte.target.value
checkboxe.target.checked
radioe.target.value (같은 name으로 묶기)
filee.target.files (비제어)
select multiple[...e.target.selectedOptions].map(o => o.value)

유효성 검사

function SignupForm() {
  const [form, setForm] = useState({ email: "", password: "" });
  const [touched, setTouched] = useState({});      // 사용자가 건드린 필드
  const [submitted, setSubmitted] = useState(false);

  // 검사 로직은 순수 함수로 분리합니다 (27장)
  const errors = {};
  if (!form.email) errors.email = "이메일을 입력해 주세요";
  else if (!form.email.includes("@")) errors.email = "이메일 형식이 아닙니다";
  if (form.password.length < 8) errors.password = "8자 이상 입력해 주세요";

  const isValid = Object.keys(errors).length === 0;

  // 에러를 '언제' 보여줄지가 사용성의 핵심입니다
  function shouldShow(field) {
    return (touched[field] || submitted) && errors[field];
  }

  return (
    <form onSubmit={handleSubmit} noValidate>
      <label htmlFor="email">이메일</label>
      <input
        id="email" name="email" type="email"
        value={form.email}
        onChange={handleChange}
        onBlur={() => setTouched(t => ({ ...t, email: true }))}
        aria-invalid={Boolean(shouldShow("email"))}
        aria-describedby={shouldShow("email") ? "email-error" : undefined}
      />
      {shouldShow("email") && (
        <p id="email-error" role="alert">{errors.email}</p>
      )}

      <button disabled={!isValid}>가입</button>
    </form>
  );
}

⏰ 에러를 보여주는 시점

  • 입력하자마자 → 두 글자 쳤는데 "8자 이상"이 떠서 짜증납니다
  • 포커스를 벗어날 때(onBlur) → 일반적으로 가장 좋습니다
  • 제출할 때 → 필수. 건드리지 않은 필드도 이때는 검사합니다
  • 이미 에러가 뜬 필드는 입력 중에도 실시간으로 검사해 고쳐지면 즉시 사라지게 합니다

♿ 접근성 — 빠뜨리기 쉬운 부분

  • label과 id 연결은 필수입니다(9·11장)
  • aria-invalid로 오류 상태를 알립니다
  • aria-describedby로 오류 메시지를 입력 칸과 연결합니다. 이게 없으면 스크린 리더 사용자는 무엇이 잘못됐는지 알 수 없습니다
  • role="alert"로 오류가 나타났을 때 즉시 읽어 주게 합니다
  • 색만으로 오류를 표시하지 않습니다(11장)

🚫 제출 버튼을 비활성화하는 것이 항상 좋지는 않습니다

비활성 버튼은 왜 못 누르는지 알려 주지 않습니다. 스크린 리더 사용자는 버튼이 있다는 것조차 놓칠 수 있습니다.
대안: 버튼은 누를 수 있게 두고, 누르면 모든 오류를 표시하고 첫 오류 필드로 포커스를 옮기는 방식이 접근성과 사용성 모두에서 낫다고 평가됩니다.

제출 처리

const [isSubmitting, setIsSubmitting] = useState(false);
const [serverError, setServerError] = useState(null);

async function handleSubmit(e) {
  e.preventDefault();              // ✅ 새로고침 방지
  setSubmitted(true);
  if (!isValid) return;
  if (isSubmitting) return;        // ✅ 중복 제출 방지

  setIsSubmitting(true);
  setServerError(null);
  try {
    const res = await fetch("/api/signup", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify(form)
    });
    if (!res.ok) throw new Error(await res.text());   // 37장
    navigate("/welcome");
  } catch (error) {
    setServerError("가입에 실패했습니다. 잠시 후 다시 시도해 주세요.");   // 33장
  } finally {
    setIsSubmitting(false);        // ✅ 성공·실패 모두 해제
  }
}

🛠️ 현장 노트 — 폼 체크리스트

  1. e.preventDefault()를 빠뜨리지 않았는가 (새로고침되며 입력이 날아갑니다)
  2. 중복 제출을 막았는가 (결제라면 두 번 결제됩니다)
  3. 로딩 상태를 finally에서 해제했는가
  4. 모든 입력에 연결된 label이 있는가
  5. 오류가 어느 필드의 문제인지 알 수 있는가
  6. Enter 키로 제출되는가 (form과 type="submit"이면 자동)
  7. 서버에서도 같은 검증을 하는가 (9장 — 클라이언트 검증은 보안이 아닙니다)
  8. 폼 안의 다른 버튼에 type="button"을 붙였는가 (9장)

📌 요약

  • 제어는 값이 상태에서, 비제어는 DOM에서 온다.
  • value만 주고 onChange가 없으면 입력이 안 된다.
  • 초기값은 undefined가 아니라 빈 문자열로 시작한다.
  • name을 상태 키와 맞추면 핸들러 하나로 모든 입력을 처리한다.
  • 체크박스는 checked, 파일은 비제어.
  • 에러는 onBlur와 제출 시에 보여주고, 고쳐지면 즉시 없앤다.
  • label·aria-invalid·aria-describedby·role="alert"로 접근성을 확보한다.
  • 중복 제출을 막고 로딩은 finally에서 해제한다.
  • 입력이 많아지면 React Hook Form + Zod.
