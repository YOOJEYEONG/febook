좁히기란

function print(value: string | number) {
  // value.toUpperCase();   ❌ number일 수도 있으므로 불가

  if (typeof value === "string") {
    value.toUpperCase();    // ✅ 여기서는 string 임이 확실합니다
  } else {
    value.toFixed(2);       // ✅ 여기서는 number
  }
}

🔬 제어 흐름 분석

TypeScript는 코드의 흐름을 따라가며 각 지점에서 변수의 타입을 다시 계산합니다. if를 통과했으면 그 조건이 참인 상태라는 것을 알고 있습니다. 그래서 특별한 문법 없이 평범한 JavaScript 조건문만으로 타입이 좁혀집니다. 이것이 TypeScript가 다른 타입 언어보다 자연스럽게 느껴지는 이유입니다.

좁히는 방법들

① typeof — 원시 타입 
function f(x: string | number | boolean | undefined) {
  if (typeof x === "string") { /* string */ }
  else if (typeof x === "number") { /* number */ }
  else if (typeof x === "boolean") { /* boolean */ }
  else { /* undefined */ }
}

// ⚠️ typeof null === "object" (25장)이므로 null은 이 방법으로 못 거릅니다
② 진리값 검사 — null·undefined 거르기 
function f(user: User | null) {
  if (!user) return;        // 이른 반환 (26장)
  user.name;                // ✅ 여기서는 User
}

// ⚠️ 주의: 0과 빈 문자열도 falsy 입니다 (25장)
function g(count: number | undefined) {
  if (!count) { /* count가 0일 때도 여기 옵니다 */ }
  if (count === undefined) { /* ✅ 정확합니다 */ }
  if (count != null) { /* ✅ null과 undefined 모두 거릅니다 */ }
}
③ instanceof — 클래스 
function handle(error: unknown) {
  if (error instanceof ValidationError) {
    error.field;            // ✅ (33장의 커스텀 에러)
  } else if (error instanceof Error) {
    error.message;
  } else {
    String(error);
  }
}
// catch 블록의 값은 unknown 입니다 (useUnknownInCatchVariables)
// 그래서 이런 확인이 반드시 필요합니다
④ in — 속성 존재 여부 
type Dog = { bark(): void };
type Cat = { meow(): void };

function speak(animal: Dog | Cat) {
  if ("bark" in animal) {
    animal.bark();          // ✅ Dog
  } else {
    animal.meow();          // ✅ Cat
  }
}
⑤ 판별 유니온 — 가장 권장되는 방법 (52장) 
type Shape =
  | { kind: "circle"; radius: number }
  | { kind: "square"; size: number }
  | { kind: "rect"; width: number; height: number };

function area(shape: Shape): number {
  switch (shape.kind) {
    case "circle": return Math.PI * shape.radius ** 2;
    case "square": return shape.size ** 2;
    case "rect":   return shape.width * shape.height;
  }
}

✅ never로 빠짐없이 처리했는지 검사하기

function area(shape: Shape): number {
  switch (shape.kind) {
    case "circle": return Math.PI * shape.radius ** 2;
    case "square": return shape.size ** 2;
    // rect를 빠뜨렸다고 합시다
    default: {
      const exhaustive: never = shape;
      // ❌ Type '{ kind: "rect"; … }' is not assignable to type 'never'
      //    → 처리하지 않은 경우가 있다고 알려 줍니다
      throw new Error(`처리되지 않은 종류: ${exhaustive}`);
    }
  }
}

나중에 새로운 종류를 추가하면 이 자리에서 컴파일 오류가 나서 "어디를 고쳐야 하는지" 알려 줍니다. 매우 강력한 패턴입니다.

사용자 정의 타입 가드

// 반환 타입에 'value is Type' 을 씁니다
function isString(value: unknown): value is string {
  return typeof value === "string";
}

function isUser(value: unknown): value is User {
  return (
    typeof value === "object" &&
    value !== null &&
    "id" in value &&
    "name" in value
  );
}

function handle(value: unknown) {
  if (isUser(value)) {
    value.name;             // ✅ User 로 좁혀집니다
  }
}

// 배열 필터링에서 특히 유용합니다
const items: (User | null)[] = [user1, null, user2];

const clean1 = items.filter(item => item !== null);
//    ↑ 여전히 (User | null)[] 로 추론될 수 있습니다

function isNotNull<T>(value: T | null): value is T {
  return value !== null;
}
const clean2 = items.filter(isNotNull);    // ✅ User[]

⚠️ 타입 가드는 거짓말할 수 있습니다

function isUser(value: unknown): value is User {
  return true;              // 😱 아무것도 검사하지 않아도 통과합니다
}

컴파일러는 여러분의 말을 믿습니다. 그래서 타입 가드 안의 검사 로직이 실제로 정확한지는 사람이 책임져야 합니다. 속성 하나만 확인하고 "이건 User다"라고 선언하면 나중에 터집니다.

타입 단언 — 최후의 수단

const input = document.getElementById("email") as HTMLInputElement;
input.value;      // ✅ 컴파일은 통과합니다

// 하지만 그 요소가 없거나 input이 아니면 런타임에 터집니다
표현의미위험도
value as Type"이 타입이라고 믿어라"중간 — 틀리면 런타임 오류
value!"null·undefined가 아니라고 믿어라"중간
value as unknown as Type"관계없는 타입으로 강제 변환"매우 높음 — 거의 항상 잘못된 설계 신호
satisfies검사하되 추론은 유지안전
as 대신 안전하게 
// ❌ 단언
const input = document.getElementById("email") as HTMLInputElement;

// ✅ 확인
const el = document.getElementById("email");
if (el instanceof HTMLInputElement) {
  el.value;
}

// ✅ 또는 옵셔널 체이닝과 함께 (26장)
const value = (document.getElementById("email") as HTMLInputElement | null)?.value;

satisfies — 검사와 추론을 동시에

type Config = Record<string, string | number>;

// ❌ 타입을 명시하면 구체적인 정보를 잃습니다
const a: Config = { port: 3000, host: "localhost" };
a.port.toFixed();     // ❌ string일 수도 있다고 판단합니다

// ✅ satisfies — Config를 만족하는지 검사하되, 추론은 구체적으로 유지
const b = { port: 3000, host: "localhost" } satisfies Config;
b.port.toFixed();     // ✅ number 임을 알고 있습니다
b.host.toUpperCase(); // ✅ string
// b.wrong;           ❌ 없는 키는 여전히 오류

satisfies는 TypeScript 4.9부터 사용할 수 있습니다. "이 값이 이 타입에 맞는지 검사해 줘. 단, 타입을 넓히지는 마"라는 뜻입니다.

외부 데이터는 런타임에 검증하세요

🚨 이 코드는 거짓말입니다

const user: User = await response.json();
// 서버가 무엇을 보내든 통과합니다. TypeScript는 확인하지 않습니다.
user.name.toUpperCase();   // name이 없으면 런타임 오류
Zod로 검증하기 — 실무 표준 
import { z } from "zod";

const UserSchema = z.object({
  id: z.number(),
  name: z.string(),
  email: z.string().email(),
  role: z.enum(["admin", "user"]),
  createdAt: z.string().datetime()
});

// 스키마에서 타입을 자동으로 뽑아냅니다 — 중복 정의가 없습니다
type User = z.infer<typeof UserSchema>;

async function getUser(id: number): Promise<User> {
  const response = await fetch(`/api/users/${id}`);
  const raw = await response.json();

  // 실제로 검사합니다. 형태가 다르면 여기서 오류가 납니다
  return UserSchema.parse(raw);
}

🛡️ 검증이 필요한 지점

  • API 응답 — 서버가 언제든 바뀔 수 있습니다
  • 사용자 입력 — 폼, 쿼리 문자열
  • localStorage에서 읽은 값 — 옛 버전의 데이터가 남아 있을 수 있습니다(39장)
  • 환경 변수 — 배포 설정 누락을 시작 시점에 잡습니다

내 코드 안은 TypeScript가 지켜 주고, 바깥과의 경계는 런타임 검증이 지킵니다. 둘 다 필요합니다.

🛠️ 현장 노트 — as가 늘어난다면

코드에 as가 많아진다면 타입 설계가 잘못되었다는 신호입니다. 잘 설계된 타입에서는 좁히기만으로 대부분 해결됩니다.

  • 유니온이 너무 넓지 않은지 — 판별 유니온으로 바꿀 수 있는지
  • 함수가 너무 많은 경우를 처리하려 하는지 — 나눌 수 있는지
  • any가 흘러들어와 그것을 as로 수습하고 있는지

as를 쓸 때는 왜 안전한지 주석으로 남기세요. "이 시점에는 반드시 로드되어 있음" 같은 근거가 있어야 다음 사람이 판단할 수 있습니다.

📌 요약

  • TypeScript는 제어 흐름을 분석해 평범한 조건문만으로 타입을 좁힌다.
  • 도구: typeof · 진리값 · instanceof · in · 판별 유니온.
  • 진리값 검사는 0과 빈 문자열도 걸러 버린다. != null이 정확하다.
  • never 검사로 모든 경우를 처리했는지 컴파일러가 확인하게 만든다.
  • 타입 가드(value is T)는 내용의 정확성을 사람이 책임진다.
  • as는 검사하지 않는다. 확인 가능한 상황이면 좁히기를 쓴다.
  • satisfies는 검사하되 추론을 유지한다.
  • 외부 데이터는 Zod 같은 도구로 런타임 검증한다.
