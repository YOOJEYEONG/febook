Context

// ① 만들기
const ThemeContext = createContext("light");

// ② 제공하기
function App() {
  const [theme, setTheme] = useState("light");
  return (
    <ThemeContext.Provider value={theme}>
      <Layout />
    </ThemeContext.Provider>
  );
}

// ③ 사용하기 — 몇 단계 아래든 상관없습니다
function DeepChild() {
  const theme = useContext(ThemeContext);
  return <div className={theme}>…</div>;
}

📦 실무에서는 훅으로 감쌉니다

const ThemeContext = createContext(null);

export function ThemeProvider({ children }) {
  const [theme, setTheme] = useState("light");
  const value = useMemo(() => ({ theme, setTheme }), [theme]);   // ⭐ 아래 설명
  return <ThemeContext.Provider value={value}>{children}</ThemeContext.Provider>;
}

export function useTheme() {
  const context = useContext(ThemeContext);
  if (context === null) {
    throw new Error("useTheme은 ThemeProvider 안에서만 쓸 수 있습니다");
  }
  return context;
}

이렇게 하면 Provider 밖에서 쓰는 실수를 즉시 알 수 있고, Context 객체를 밖으로 노출하지 않아도 됩니다.

성능 함정

💥 value 객체를 매번 새로 만들면

// ❌ 렌더링마다 새 객체 → 모든 소비자가 다시 렌더링됩니다
<AuthContext.Provider value={{ user, login, logout }}>

// ✅ useMemo로 참조를 안정시킵니다 (64장)
const value = useMemo(() => ({ user, login, logout }), [user, login, logout]);
<AuthContext.Provider value={value}>

🌊 Context는 부분 구독을 지원하지 않습니다

하나의 Context에 user·theme·cart를 전부 담으면, 어느 하나만 바뀌어도 그 Context를 쓰는 모든 컴포넌트가 다시 렌더링됩니다. theme만 쓰는 컴포넌트도 cart가 바뀌면 렌더링됩니다.

  • 해결 1 — Context를 목적별로 나눕니다. ThemeContext, AuthContext 따로.
  • 해결 2 — 값과 갱신 함수를 분리합니다. 값은 자주 바뀌지만 함수는 안 바뀝니다.
  • 해결 3 — 자주 바뀌는 상태는 전역 상태 라이브러리를 씁니다. 대부분 부분 구독을 지원합니다.

🎯 Context에 어울리는 것

  • 자주 바뀌지 않고 많은 곳에서 필요한 값
  • 테마, 로그인 사용자 정보, 언어 설정, 라우터 정보
  • 컴파운드 컴포넌트의 내부 상태(63장)

어울리지 않는 것: 입력 중인 폼 값, 마우스 위치, 초당 여러 번 바뀌는 값.

useReducer

// 상태 변경 로직을 컴포넌트 밖의 순수 함수로 (27장)
function reducer(state, action) {
  switch (action.type) {
    case "added":
      return { ...state, items: [...state.items, action.item] };
    case "removed":
      return { ...state, items: state.items.filter(i => i.id !== action.id) };
    case "cleared":
      return { ...state, items: [] };
    default:
      throw new Error("알 수 없는 action: " + action.type);
  }
}

function Cart() {
  const [state, dispatch] = useReducer(reducer, { items: [] });

  return <button onClick={() => dispatch({ type: "added", item })}>담기</button>;
}

가장 중요한 구분: 서버 상태와 클라이언트 상태

🔀 이 구분을 놓치면 전역 상태 관리가 지옥이 됩니다

서버 상태클라이언트 상태
예사용자 목록, 상품 정보, 주문 내역모달 열림, 선택된 탭, 다크 모드, 폼 입력값

서버 데이터를 Redux 같은 곳에 넣고 직접 관리하려는 것이 과거 프론트엔드가 복잡해진 큰 이유였습니다. 지금은 서버 상태 전용 도구가 그 일을 대신하고, 그 결과 전역 상태 라이브러리에 남는 것은 놀랄 만큼 적어집니다.

전역 상태 라이브러리

도구특징
Zustand매우 단순. 보일러플레이트가 거의 없고 부분 구독을 지원합니다. 현재 가장 인기 있는 선택
Redux Toolkit큰 팀·복잡한 앱. DevTools의 시간 여행 디버깅이 강력합니다. 규칙이 엄격해 일관성이 좋습니다
Jotai / Recoil작은 단위(atom)로 상태를 쪼개는 방식. 세밀한 구독
Context + useReducer추가 의존성 없음. 소규모에 충분합니다
Zustand의 느낌 
import { create } from "zustand";

const useCartStore = create((set) => ({
  items: [],
  add: (item) => set((s) => ({ items: [...s.items, item] })),
  clear: () => set({ items: [] })
}));

function Cart() {
  // ⭐ 필요한 것만 구독합니다 — items가 바뀌어도 count만 쓰는 컴포넌트는 안 바뀝니다
  const items = useCartStore((s) => s.items);
  const add = useCartStore((s) => s.add);
}

🛠️ 현장 노트 — 선택 순서

  1. 먼저 useState로 시작합니다. 대부분의 상태는 한 컴포넌트 안에 있으면 됩니다.
  2. 두세 컴포넌트가 공유하면 상태 끌어올리기(59장).
  3. props 전달이 길어지면 합성으로 해결을 먼저 시도(63장).
  4. 정말 넓은 범위면 Context.
  5. Context로 성능이 문제되거나 로직이 복잡하면 Zustand 같은 라이브러리.
  6. 서버 데이터는 처음부터 TanStack Query(67장). 여기에 넣지 마세요.

"React 프로젝트니까 상태 관리 라이브러리를 깔아야지"는 잘못된 출발입니다. 서버 상태를 분리하고 나면 전역으로 관리할 것이 거의 없다는 것을 알게 되는 경우가 많습니다.

URL도 상태 저장소입니다

// ❌ 검색어·필터·페이지를 상태로만 관리하면
// 새로고침하면 사라지고, 링크를 공유할 수 없고, 뒤로 가기가 동작하지 않습니다
const [page, setPage] = useState(1);

// ✅ URL에 담습니다 (3·40장)
const [searchParams, setSearchParams] = useSearchParams();
const page = Number(searchParams.get("page") ?? 1);

공유 가능하고 뒤로 가기가 동작해야 하는 것은 URL에 두는 것이 맞습니다. 검색어, 필터, 정렬, 페이지, 열린 탭이 대표적입니다. 68장에서 다시 다룹니다.

📌 요약

  • Context는 깊은 props 전달을 없애 준다. 커스텀 훅으로 감싸 쓴다.
  • value 객체는 useMemo로 안정화한다.
  • Context는 부분 구독을 지원하지 않는다. 목적별로 나눈다.
  • Context에 어울리는 것은 자주 바뀌지 않는 넓은 범위의 값이다.
  • useReducer는 상태 변경이 여러 방식일 때, 그리고 테스트하고 싶을 때.
  • 서버 상태와 클라이언트 상태를 구분하는 것이 가장 중요하다.
  • 서버 데이터는 TanStack Query, 클라이언트 상태만 전역 라이브러리로.
  • 공유·뒤로 가기가 필요한 상태는 URL에 둔다.
  • 라이브러리부터 깔지 말고 필요해질 때 도입한다.
