함수를 만드는 세 가지 방법

// ① 함수 선언문
function add(a, b) {
  return a + b;
}

// ② 함수 표현식 — 값으로 만들어 변수에 담는다
const subtract = function (a, b) {
  return a - b;
};

// ③ 화살표 함수
const multiply = (a, b) => a * b;

화살표 함수 문법

// 매개변수가 여러 개
const add = (a, b) => a + b;

// 하나면 괄호 생략 가능 (팀에 따라 항상 붙이기도 합니다)
const double = n => n * 2;

// 없으면 빈 괄호
const hello = () => "안녕";

// 본문이 한 줄이면 return 생략 (암묵적 반환)
const square = n => n * n;

// 여러 줄이면 중괄호와 return 필요
const process = n => {
  const doubled = n * 2;
  return doubled + 1;
};

// 객체를 반환할 때는 괄호로 감싸야 합니다
const makeUser = name => ({ name: name });
//                       ↑ 이 괄호가 없으면 { } 를 함수 본문으로 해석합니다

🔺 화살표 함수를 쓰면 안 되는 곳

// ❌ 객체의 메서드 — this가 객체를 가리키지 않습니다 (30장)
const counter = {
  count: 0,
  increase: () => { this.count++; }   // this가 엉뚱한 곳
};

// ✅ 메서드 축약 문법
const counter2 = {
  count: 0,
  increase() { this.count++; }
};

// ❌ DOM 이벤트 핸들러에서 this로 요소를 가리키려 할 때
button.addEventListener("click", () => {
  this.classList.add("on");   // this가 요소가 아님
});

// ✅ event.currentTarget 을 쓰면 화살표 함수여도 안전합니다
button.addEventListener("click", (event) => {
  event.currentTarget.classList.add("on");
});

반환

function f1() { return 1; }
function f2() { }                  // return 없음 → undefined 반환
function f3() { return; }          // 값 없는 return → undefined

// return 이후의 코드는 실행되지 않습니다
function f4() {
  return "끝";
  console.log("실행 안 됨");
}

// ⚠️ 함정: return 뒤에서 줄을 바꾸지 마세요
function bad() {
  return
    { name: "홍길동" };   // 자동 세미콜론 삽입으로 return; 이 되어 undefined
}

값을 여러 개 돌려주고 싶으면 객체나 배열로 묶습니다.

function getMinMax(numbers) {
  return { min: Math.min(...numbers), max: Math.max(...numbers) };
}
const { min, max } = getMinMax([3, 1, 4, 1, 5]);   // 구조 분해 (28장)

함수는 값이다

JavaScript에서 함수는 일급 객체입니다. 숫자나 문자열처럼 변수에 담고, 인자로 넘기고, 반환할 수 있습니다.

💥 신입이 가장 많이 하는 실수

// ❌ 지금 즉시 실행되고, 그 결과(undefined)가 등록됩니다
button.addEventListener("click", handleClick());

// ✅ 함수 자체를 넘깁니다. 클릭할 때 실행됩니다
button.addEventListener("click", handleClick);

// ✅ 인자를 넘겨야 한다면 감싸세요
button.addEventListener("click", () => handleClick(id));

괄호를 붙이면 "지금 실행", 안 붙이면 "함수 그 자체"입니다.

좋은 함수의 조건

순수 함수

순수 함수는 두 조건을 만족합니다.

  1. 같은 입력이면 항상 같은 출력.
  2. 바깥 세계를 바꾸지 않는다 (부수 효과 없음).
// ✅ 순수 함수 — 예측 가능하고 테스트하기 쉽습니다
function add(a, b) {
  return a + b;
}

// ❌ 순수하지 않음: 바깥 변수를 바꿉니다
let total = 0;
function addToTotal(n) {
  total += n;
}

// ❌ 순수하지 않음: 인자로 받은 배열을 직접 수정합니다
function addItemBad(list, item) {
  list.push(item);        // 원본이 바뀝니다
  return list;
}

// ✅ 새 배열을 만들어 반환합니다
function addItem(list, item) {
  return [...list, item];
}

모든 함수가 순수할 수는 없습니다. 화면을 바꾸고 서버에 요청하는 일은 본질적으로 부수 효과입니다. 요령은 순수한 계산 부분과 부수 효과 부분을 분리하는 것입니다. React가 이 원칙 위에 세워져 있어서(57장), 지금 익혀 두면 나중이 훨씬 수월합니다.

🛠️ 현장 노트 — 함수 작성 원칙 5가지

  1. 하나의 함수는 하나의 일만. 이름에 "and"가 들어가면 쪼갤 신호입니다.
  2. 이름은 동사로. getUserName, calculateTotal, isValid.
  3. 인자는 3개 이하. 넘으면 객체로 받습니다.
  4. 이른 반환으로 중첩을 줄입니다.
  5. 인자로 받은 객체·배열을 직접 수정하지 마세요. 호출한 쪽이 예상하지 못합니다.

📌 요약

  • 함수 선언문은 통째로 호이스팅되어 위에서 호출할 수 있다. 표현식·화살표는 아니다.
  • 화살표 함수는 this·arguments가 없고 new로 호출할 수 없다.
  • 객체 메서드에는 메서드 축약 문법을 쓴다. 화살표 함수는 부적절하다.
  • 화살표 함수로 객체를 반환할 때는 ({ … })로 감싼다.
  • 기본값·나머지 매개변수·객체 구조 분해로 인자를 유연하게 받는다.
  • 함수는 값이다. 변수에 담고, 넘기고, 반환할 수 있다.
  • fn은 함수 자체, fn()은 실행 결과. 콜백에 괄호를 붙이지 않는다.
  • 가능하면 순수 함수로. 부수 효과는 분리한다.
