동적 메타데이터 — 데이터에 따라 달라질 때 
// app/blog/[slug]/page.tsx
export async function generateMetadata({ params }) {
  const { slug } = await params;
  const post = await getPost(slug);

  if (!post) return { title: "글을 찾을 수 없습니다" };

  return {
    title: post.title,
    description: post.summary,
    openGraph: {
      title: post.title,
      images: [post.coverImage],
      type: "article",
      publishedTime: post.publishedAt
    }
  };
}

🖼️ 미리보기 이미지를 코드로 만들기

// app/blog/[slug]/opengraph-image.tsx
import { ImageResponse } from "next/og";

export default async function Image({ params }) {
  const post = await getPost(params.slug);
  return new ImageResponse(
    (
      <div style={{ display: "flex", fontSize: 60, background: "white", padding: 60 }}>
        {post.title}
      </div>
    ),
    { width: 1200, height: 630 }
  );
}

글마다 제목이 들어간 미리보기 이미지를 자동 생성합니다. 디자이너가 매번 이미지를 만들 필요가 없습니다.

사이트맵과 robots.txt (10장) 
// app/sitemap.ts
export default async function sitemap() {
  const posts = await getAllPosts();
  return [
    { url: "https://example.com", lastModified: new Date(), priority: 1 },
    ...posts.map(p => ({
      url: `https://example.com/blog/${p.slug}`,
      lastModified: p.updatedAt
    }))
  ];
}

// app/robots.ts
export default function robots() {
  return {
    rules: { userAgent: "*", allow: "/", disallow: "/admin/" },
    sitemap: "https://example.com/sitemap.xml"
  };
}

이미지 최적화

import Image from "next/image";

<Image
  src="/hero.jpg"
  alt="바다가 보이는 창가"      // 8장 — 필수입니다
  width={1200}
  height={630}
  priority                        // 첫 화면 이미지에만
/>
자동으로 해 주는 것직접 하면
WebP·AVIF 변환이미지마다 여러 포맷을 만들고 <picture>를 써야 합니다(8장)
크기별 생성 + srcset손으로 리사이즈하고 srcset·sizes를 작성
지연 로딩loading="lazy"를 빠뜨리기 쉽습니다
레이아웃 이동 방지width·height로 공간 확보(8장 CLS)
흐릿한 자리 표시자별도 구현 필요

⭐ priority를 정확히 쓰세요

next/image는 기본이 지연 로딩입니다. 그래서 첫 화면의 큰 대표 이미지에 priority를 안 붙이면 그 이미지가 늦게 로드되어 LCP 점수가 나빠집니다(8·76장).
반대로 모든 이미지에 붙이면 우선순위 개념이 사라져 아무 의미가 없어집니다. 화면에 처음부터 보이는 큰 이미지 한두 개에만 붙이세요.

🔒 왜 도메인을 허용해야 하나

아무 주소나 허용하면 남이 여러분의 서버를 이미지 변환 서버로 악용할 수 있습니다. 비용이 청구되는 것은 여러분입니다. 그래서 명시적으로 허용한 도메인만 처리합니다.

폰트 최적화

// app/layout.tsx
import localFont from "next/font/local";

const pretendard = localFont({
  src: "./fonts/PretendardVariable.woff2",
  display: "swap",              // 19장
  variable: "--font-pretendard",
  preload: true
});

export default function RootLayout({ children }) {
  return (
    <html lang="ko" className={pretendard.variable}>
      <body>{children}</body>
    </html>
  );
}
해 주는 것왜 중요한가
자체 호스팅외부 폰트 서버로의 추가 연결이 사라집니다. 개인정보·성능 모두 이득
preload 자동crossorigin을 빠뜨리는 실수를 막아 줍니다(19장)
대체 폰트 자동 조정폰트 교체 시 레이아웃이 튀지 않습니다(19장의 size-adjust)
빌드 시점 처리실행 중에 계산하지 않습니다

🇰🇷 한글 폰트는 서브셋이 여전히 중요합니다

19장에서 말했듯 한글 폰트는 수 MB에 이릅니다. next/font가 자동으로 해 주지 않는 부분이 바로 서브셋입니다.
가변 폰트를 쓰거나, unicode-range로 나뉜 파일을 쓰거나, 필요한 글자만 남긴 서브셋 파일을 준비해야 합니다. 폰트 하나가 페이지에서 가장 무거운 자원이 되는 일이 흔합니다.

🐌 서드파티 스크립트가 성능의 최대 적입니다

여러분의 코드를 아무리 최적화해도 채팅 위젯 하나가 500KB를 가져오면 소용없습니다. 그리고 그 스크립트는 여러분이 통제할 수 없습니다.
필요성을 다시 검토하고, 꼭 필요하다면 가능한 한 늦게 로드하세요. 성능 문제를 조사할 때 남의 스크립트부터 확인하는 것이 요령입니다(76장).

동적 임포트

import dynamic from "next/dynamic";

// 무거운 컴포넌트를 필요할 때만 (34·47·68장)
const Editor = dynamic(() => import("./Editor"), {
  loading: () => <EditorSkeleton />,
  ssr: false            // 브라우저 API에 의존하는 컴포넌트라면
});

ssr: false는 서버에서 렌더링하지 않겠다는 뜻입니다. window를 직접 쓰는 지도·차트 라이브러리에서 하이드레이션 불일치(70장)를 피할 때 유용합니다.

🛠️ 현장 노트 — 배포 전 확인 목록

  1. 모든 페이지에 고유한 title·description이 있는가(10장)
  2. OG 이미지가 절대 주소이고 실제로 표시되는가
  3. 첫 화면 큰 이미지에 priority가 붙어 있는가
  4. 모든 이미지에 alt가 있는가(8장)
  5. 폰트 용량을 확인했는가 (한글은 특히)
  6. 서드파티 스크립트를 꼭 필요한 것만 남겼는가
  7. npm run build 출력에서 비정상적으로 큰 라우트가 없는가(47장)

📌 요약

  • metadata와 generateMetadata로 페이지별 SEO·공유 미리보기를 만든다.
  • sitemap.ts·robots.ts로 크롤러에게 구조를 알린다.
  • next/image가 포맷 변환·크기별 생성·지연 로딩·CLS 방지를 자동으로 한다.
  • 첫 화면 큰 이미지에만 priority. 전부에 붙이면 의미가 없다.
  • 외부 이미지는 도메인 허용이 필요하다 (악용 방지).
  • next/font는 자체 호스팅·프리로드·대체 폰트 조정을 해 주지만 한글 서브셋은 직접 해야 한다.
  • 외부 스크립트는 가능한 한 늦게. 성능의 최대 적이다.
  • 브라우저 API에 의존하는 컴포넌트는 ssr: false로 하이드레이션 문제를 피한다.
