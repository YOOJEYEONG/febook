JSX는 함수 호출입니다

// 우리가 쓰는 것
const element = <h1 className="title">안녕하세요</h1>;

// 빌드 도구가 바꾸는 것 (React 17 이후의 새 변환 방식)
const element = _jsx("h1", { className: "title", children: "안녕하세요" });

JSX는 HTML이 아닙니다. class="term" data-term="javascript">JavaScript 안에 UI 구조를 편하게 쓰기 위한 문법 확장이고, 빌드 시점에 평범한 함수 호출로 바뀝니다. 그래서 JSX는 값입니다.

// 값이므로 변수에 담고, 배열에 넣고, 반환할 수 있습니다
const greeting = <p>안녕</p>;
const items = [<li key="1">A</li>, <li key="2">B</li>];
const message = isLoggedIn ? <Dashboard /> : <Login />;

중괄호로 JavaScript 넣기 
function Profile({ user }) {
  const isAdmin = user.role === "admin";

  return (
    <div className={isAdmin ? "card admin" : "card"}>
      <h2>{user.name}</h2>
      <p>나이: {user.age}세</p>
      <p>가입: {user.createdAt.toLocaleDateString("ko-KR")}</p>
      <img src={user.avatarUrl} alt={`${user.name}의 프로필 사진`} />
      <p style={{ color: isAdmin ? "crimson" : "inherit", fontWeight: 700 }}>
        {isAdmin ? "관리자" : "일반 회원"}
      </p>
    </div>
  );
}

🎯 중괄호 안에는 표현식만

{user.name}                       ✅ 값
{count + 1}                       ✅ 계산
{items.map(i => <li>{i}</li>)}    ✅ 함수 호출
{isOk ? "예" : "아니오"}           ✅ 삼항 연산자

{if (isOk) { … }}                 ❌ if는 '문'이라 값이 아닙니다
{for (…) { … }}                   ❌ for도 마찬가지

조건 분기가 복잡하면 JSX 밖에서 변수로 계산한 뒤 넣으세요. 26장에서 배운 "값을 고를 때는 삼항, 동작을 나눌 때는 if"가 그대로 적용됩니다.

최상위는 하나여야 합니다

// ❌ 형제가 둘
return (
  <h1>제목</h1>
  <p>내용</p>
);

// ✅ 감싸기
return (
  <div>
    <h1>제목</h1>
    <p>내용</p>
  </div>
);

// ✅ 불필요한 div를 만들고 싶지 않다면 프래그먼트
return (
  <>
    <h1>제목</h1>
    <p>내용</p>
  </>
);

함수는 값을 하나만 반환할 수 있기 때문입니다. 프래그먼트(<>…</>)는 실제 DOM에 아무 요소도 만들지 않습니다. 표(<tbody> 안)나 flex·grid 자식처럼 중간에 div가 끼면 안 되는 곳에 필수입니다. key가 필요하면 <React.Fragment key={id}>로 씁니다.

컴포넌트

// 컴포넌트 = JSX를 반환하는 함수. 이름은 반드시 대문자로 시작
function Welcome() {
  return <h1>환영합니다</h1>;
}

// 사용
<Welcome />

🔠 대문자로 시작해야 합니다

<welcome />    // → "welcome"이라는 HTML 태그로 해석됩니다
<Welcome />    // → Welcome 이라는 컴포넌트 함수 호출

React는 첫 글자로 구분합니다. 소문자면 HTML 태그, 대문자면 컴포넌트입니다. "컴포넌트를 넣었는데 아무것도 안 나온다"면 가장 먼저 확인할 부분입니다.

props

// 넘기는 쪽
<UserCard name="홍길동" age={20} isAdmin tags={["신규", "VIP"]} />

// 받는 쪽 — props는 하나의 객체로 들어옵니다
function UserCard(props) {
  return <h2>{props.name}</h2>;
}

// ✅ 구조 분해로 받는 것이 표준입니다 (28장)
function UserCard({ name, age, isAdmin = false, tags = [] }) {
  return (
    <div>
      <h2>{name} {isAdmin && "👑"}</h2>
      <p>{age}세</p>
      <p>{tags.join(", ")}</p>
    </div>
  );
}
전달 방법의미
name="홍길동"문자열은 따옴표로 그대로
age={20}숫자·불리언·객체·배열·함수는 중괄호
isAdmin값 없이 쓰면 true (HTML 불리언 속성과 같습니다)
onClick={handleClick}함수 자체를 넘깁니다. handleClick()이 아닙니다(27장)
{...props}객체를 통째로 펼쳐 전달

🔒 props는 읽기 전용입니다

function Bad({ user }) {
  user.name = "변경";        // ❌ 절대 하지 마세요
  return <p>{user.name}</p>;
}

props를 바꾸면 React는 그 사실을 모릅니다. 화면이 갱신되지 않고, 부모의 데이터까지 오염되어(25장 참조 복사) 추적이 매우 어려운 버그가 됩니다.
데이터는 위에서 아래로만 흐릅니다. 자식이 무언가를 바꿔야 한다면 부모가 넘겨준 함수를 호출해서 부모가 바꾸게 해야 합니다.

자식이 부모에게 알리는 법 
function Parent() {
  const [count, setCount] = useState(0);

  return (
    <Child
      count={count}
      onIncrease={() => setCount(count + 1)}   // ← 함수를 내려보냅니다
    />
  );
}

function Child({ count, onIncrease }) {
  return <button onClick={onIncrease}>{count}</button>;
  //     자식은 "눌렸다"고 알릴 뿐, 상태는 부모가 관리합니다
}

children

function Card({ title, children }) {
  return (
    <section className="card">
      <h2>{title}</h2>
      <div className="card-body">{children}</div>
    </section>
  );
}

// 사용 — 태그 사이의 내용이 children으로 들어갑니다
<Card title="공지">
  <p>내용이 여기 들어갑니다</p>
  <button>확인</button>
</Card>

🎁 children이 있으면 컴포넌트가 유연해집니다

<Card content="..." />처럼 props로만 받으면 넣을 수 있는 것이 제한됩니다. children을 받으면 무엇이든 넣을 수 있습니다. 레이아웃·모달·패널처럼 껍데기 역할을 하는 컴포넌트에 특히 유용합니다. 63장에서 더 다룹니다.

조합하기

function App() {
  const users = [
    { id: 1, name: "홍길동", age: 20 },
    { id: 2, name: "김철수", age: 30 }
  ];

  return (
    <Layout>
      <Header title="사용자 목록" />
      <main>
        {users.map(user => (
          <UserCard key={user.id} {...user} />
        ))}
      </main>
      <Footer />
    </Layout>
  );
}

🚫 컴포넌트 안에 컴포넌트를 정의하지 마세요

// ❌ 매 렌더링마다 '새로운 함수'가 만들어집니다
function Parent() {
  function Child() { return <p>안녕</p>; }
  return <Child />;
}
// React는 이것을 '다른 컴포넌트'로 보고 매번 새로 만듭니다.
// → 상태가 초기화되고, 입력 포커스가 사라지고, 성능이 나빠집니다.

// ✅ 밖에 정의합니다
function Child() { return <p>안녕</p>; }
function Parent() { return <Child />; }

컴포넌트를 언제 나눌까

나눌 신호설명
같은 UI가 반복된다가장 명확한 신호
이름을 붙일 수 있다"이 부분은 사용자 카드다"라고 말할 수 있으면 컴포넌트
한 화면에 스크롤이 필요할 만큼 길다대략 100~150줄이 넘으면 검토
독립적으로 테스트하고 싶다—

🛠️ 현장 노트 — 너무 잘게 쪼개지도 마세요

<Title>, <Subtitle>, <Spacer>처럼 한 줄짜리 컴포넌트를 남발하면 파일을 계속 오가야 해서 오히려 읽기 어려워집니다.
좋은 기준은 "이 컴포넌트를 한 문장으로 설명할 수 있는가"입니다. "제목과 내용을 받아 카드로 보여준다"는 좋고, "h2를 감싼다"는 나쁩니다.
그리고 처음부터 완벽하게 나누려 하지 마세요. 한 파일에 크게 쓴 뒤, 반복이 보이거나 길어질 때 나누는 것이 실무의 실제 순서입니다.

📌 요약

  • JSX는 HTML이 아니라 함수 호출로 변환되는 문법이다. 그래서 값이다.
  • className·htmlFor·카멜케이스·style={{}}·자기 닫기 태그를 기억한다.
  • 중괄호에는 표현식만. if·for는 밖에서 처리한다.
  • 반환값은 하나여야 한다. 필요하면 프래그먼트.
  • 컴포넌트 이름은 대문자로 시작해야 한다.
  • props는 읽기 전용이다. 데이터는 위에서 아래로, 변경 요청은 함수로 위로.
  • children으로 무엇이든 감쌀 수 있는 컴포넌트를 만든다.
  • 컴포넌트 안에 컴포넌트를 정의하지 않는다.
