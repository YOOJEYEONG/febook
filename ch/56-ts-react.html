function Form() {
  // 인라인이면 타입이 자동 추론됩니다 — 이게 가장 편합니다

}

💡 props로 받는 핸들러 타입

interface Props {
  onClick: () => void;                                    // 인자 없음 — 가장 흔함
  onChange: (value: string) => void;                      // 가공된 값을 넘길 때

가능하면 이벤트 객체를 그대로 넘기지 마세요. onChange: (value: string) => void처럼 의미 있는 값을 넘기면 컴포넌트를 쓰는 쪽이 훨씬 편하고, 나중에 내부 구현을 바꾸기도 쉽습니다.

useReducer — 판별 유니온과 함께 (52장) 
type State = { count: number; step: number };

type Action =
  | { type: "increment" }
  | { type: "decrement" }
  | { type: "setStep"; step: number }
  | { type: "reset" };

function reducer(state: State, action: Action): State {
  switch (action.type) {
    case "increment": return { ...state, count: state.count + state.step };
    case "decrement": return { ...state, count: state.count - state.step };
    case "setStep":   return { ...state, step: action.step };   // ✅ step 존재 보장
    case "reset":     return { count: 0, step: 1 };
  }
}

const [state, dispatch] = useReducer(reducer, { count: 0, step: 1 });
dispatch({ type: "setStep", step: 5 });    // ✅
dispatch({ type: "setStep" });              // ❌ step 누락을 잡아 줍니다

function useToggle(initial = false) {

제네릭 컴포넌트

interface SelectProps<T> {
  options: T[];
  value: T;
  onChange: (value: T) => void;
  getLabel: (option: T) => string;
  getKey: (option: T) => string;
}

function Select<T>({ options, value, onChange, getLabel, getKey }: SelectProps<T>) {
  return (
    <select
      value={getKey(value)}
      onChange={(e) => {
        const found = options.find(o => getKey(o) === e.target.value);
        if (found) onChange(found);
      }}
    >
      {options.map(option => (
        <option key={getKey(option)} value={getKey(option)}>
          {getLabel(option)}
        </option>
      ))}
    </select>
  );
}

// 사용할 때 T가 자동으로 추론됩니다

🛠️ 현장 노트

  1. 대부분은 추론에 맡기세요. React + TypeScript는 추론이 꽤 좋습니다. useState·이벤트 핸들러·map 콜백은 대개 안 써도 됩니다.
  2. props 타입은 반드시 씁니다. 컴포넌트의 사용 설명서 역할을 합니다.
  3. 리터럴 유니온을 적극 쓰세요. variant: string보다 variant: "primary" | "secondary"가 훨씬 유용합니다.
  4. 막히면 React TypeScript Cheatsheet를 보세요. 거의 모든 상황의 예제가 정리되어 있습니다.
  5. React 19부터 forwardRef가 필요 없어졌습니다. 함수 컴포넌트가 ref를 일반 prop으로 받을 수 있습니다. 옛 자료에서 forwardRef를 보면 이 변화를 떠올리세요.

📌 요약

  • JSX가 있는 파일은 .tsx.
  • React.FC 대신 평범한 함수 선언을 쓴다.
  • children은 React.ReactNode.
  • 인라인 이벤트 핸들러는 타입이 자동 추론된다. 밖으로 빼면 명시한다.
  • useState([])·useState(null)은 제네릭을 명시해야 한다.
  • 커스텀 훅이 배열을 반환하면 as const를 붙인다.
  • ComponentProps<"button"> 확장으로 HTML 속성을 전부 지원한다.
  • props 타입은 컴포넌트의 문서다. 리터럴 유니온을 적극 활용한다.
