함수란 — 이름 붙인 레시피, 자판기

같은 일을 여러 번 해야 할 때마다 코드를 복사해 붙이면, 고칠 때 모든 복사본을 찾아 고쳐야 합니다. 그래서 여러 줄의 동작을 한 번 묶어 이름을 붙여 두고, 필요할 때 이름으로 불러 쓰는 것이 함수입니다.

자판기를 떠올려 보세요. 돈과 버튼 번호를 넣으면(입력) 안에서 정해진 일을 하고 음료를 내줍니다(출력). 안이 어떻게 생겼는지 몰라도, 무엇을 넣으면 무엇이 나오는지만 알면 쓸 수 있습니다.

자판기와 함수 
            ┌──────────────── 함수 add ────────────────┐
 입력 2, 3 → │ 투입구: (a, b)   안에서: a + b 계산       │ → 출력 5
            │ 매개변수         본문                    │   return
            └─────────────────────────────────────────┘
// ① 정의 — 자판기를 설치한다 (아직 아무 일도 안 일어남)
function add(a, b) {   // a, b: 매개변수 = 투입구
  return a + b;        // return: 결과를 내보내는 출구
}

// ② 호출 — 자판기에 값을 넣고 버튼을 누른다
const result = add(2, 3);   // 2, 3: 인자 = 실제로 넣는 값
console.log(result);         // 5
  • 정의와 호출은 다릅니다. 정의만 하면 아무 일도 일어나지 않고, 이름()처럼 괄호를 붙여 호출해야 실행됩니다.
  • 매개변수(parameter)는 함수 쪽의 빈칸 이름, 인자(argument)는 호출할 때 실제로 넣는 값입니다.
  • return은 결과를 호출한 쪽으로 돌려줍니다. 돌려받은 값은 변수에 담거나 바로 출력할 수 있습니다.

함수를 만드는 세 가지 방법

// ① 함수 선언문
function add(a, b) {
  return a + b;
}

// ② 함수 표현식 — 값으로 만들어 변수에 담는다
const subtract = function (a, b) {
  return a - b;
};

// ③ 화살표 함수
const multiply = (a, b) => a * b;

지금 당장 중요한 차이는 표의 첫 줄(호이스팅)뿐입니다. 나머지 줄은 각각 해당 챕터에서 배운 뒤 돌아와 보면 됩니다.

화살표 함수 문법

// 매개변수가 여러 개
const add = (a, b) => a + b;

// 하나면 괄호 생략 가능 (팀에 따라 항상 붙이기도 합니다)
const double = n => n * 2;

// 없으면 빈 괄호
const hello = () => "안녕";

// 본문이 한 줄이면 return 생략 (암묵적 반환)
const square = n => n * n;

// 여러 줄이면 중괄호와 return 필요
const process = n => {
  const doubled = n * 2;
  return doubled + 1;
};

// 객체를 반환할 때는 괄호로 감싸야 합니다
const makeUser = name => ({ name: name });
//                       ↑ 이 괄호가 없으면 { } 를 함수 본문으로 해석합니다

🔺 화살표 함수를 쓰면 안 되는 곳

// ❌ 객체의 메서드 — this가 객체를 가리키지 않습니다 (30장)
const counter = {
  count: 0,
  increase: () => { this.count++; }   // this가 엉뚱한 곳
};

// ✅ 메서드 축약 문법
const counter2 = {
  count: 0,
  increase() { this.count++; }
};

// ❌ DOM 이벤트 핸들러에서 this로 요소를 가리키려 할 때
//    (addEventListener·event.currentTarget은 38장에서 배웁니다.
//     "버튼을 클릭하면 이 함수를 실행해 줘"라는 등록 코드입니다)
button.addEventListener("click", () => {
  this.classList.add("on");   // this가 요소가 아님
});

// ✅ event.currentTarget 을 쓰면 화살표 함수여도 안전합니다
button.addEventListener("click", (event) => {
  event.currentTarget.classList.add("on");
});

매개변수

🏷️ 인자가 3개를 넘으면 객체로 받으세요

// ❌ 호출부만 보면 무슨 값인지 알 수 없습니다
createUser("홍길동", 20, "서울", true, false);

// ✅ 스스로 설명합니다. 순서도 자유롭습니다
createUser({
  name: "홍길동",
  age: 20,
  city: "서울",
  isAdmin: true,
  sendEmail: false
});

반환

function f1() { return 1; }
function f2() { }                  // return 없음 → undefined 반환
function f3() { return; }          // 값 없는 return → undefined

// return 이후의 코드는 실행되지 않습니다
function f4() {
  return "끝";
  console.log("실행 안 됨");
}

// ⚠️ 함정: return 뒤에서 줄을 바꾸지 마세요
function bad() {
  return
    { name: "홍길동" };   // 자동 세미콜론 삽입으로 return; 이 되어 undefined
}

값을 여러 개 돌려주고 싶으면 객체나 배열로 묶습니다.

function getMinMax(numbers) {
  // ...numbers: 배열을 낱개 값으로 펼쳐 넘깁니다 (전개 구문, 28장)
  return { min: Math.min(...numbers), max: Math.max(...numbers) };
}
const { min, max } = getMinMax([3, 1, 4, 1, 5]);   // 구조 분해 (28장)

함수는 값이다

JavaScript에서 함수는 일급 객체(first-class)입니다. 어려운 말 같지만 뜻은 단순합니다. "숫자·문자열 같은 다른 값과 똑같은 대우를 받는다"는 뜻입니다. 그래서 함수도 변수에 담고, 다른 함수에 인자로 넘기고, 함수의 결과로 반환할 수 있습니다.

자판기 비유로 말하면, 자판기 자체를 트럭에 실어 다른 곳에 보내거나(인자로 넘기기), 자판기 공장이 새 자판기를 만들어 내보내는(반환하기) 것도 가능하다는 뜻입니다. 다른 함수에 넘겨져 나중에 불리는 함수를 콜백(callback), 함수를 받거나 돌려주는 함수를 고차 함수라고 부릅니다.

💥 신입이 가장 많이 하는 실수

// (addEventListener는 38장. "클릭되면 이 함수를 실행해 줘"라는 등록입니다)
// ❌ 지금 즉시 실행되고, 그 결과(undefined)가 등록됩니다
button.addEventListener("click", handleClick());

// ✅ 함수 자체를 넘깁니다. 클릭할 때 실행됩니다
button.addEventListener("click", handleClick);

// ✅ 인자를 넘겨야 한다면 감싸세요
button.addEventListener("click", () => handleClick(id));

괄호를 붙이면 "지금 실행", 안 붙이면 "함수 그 자체"입니다.

좋은 함수의 조건

순수 함수

순수 함수는 두 조건을 만족합니다.

  1. 같은 입력이면 항상 같은 출력.
  2. 바깥 세계를 바꾸지 않는다 (부수 효과 없음).
// ✅ 순수 함수 — 예측 가능하고 테스트하기 쉽습니다
function add(a, b) {
  return a + b;
}

// ❌ 순수하지 않음: 바깥 변수를 바꿉니다
let total = 0;
function addToTotal(n) {
  total += n;
}

// ❌ 순수하지 않음: 인자로 받은 배열을 직접 수정합니다
function addItemBad(list, item) {
  list.push(item);        // 원본이 바뀝니다
  return list;
}

// ✅ 새 배열을 만들어 반환합니다
function addItem(list, item) {
  return [...list, item];   // 기존 값을 펼쳐 담은 새 배열 (전개 구문, 28장)
}

모든 함수가 순수할 수는 없습니다. 화면을 바꾸고 서버에 요청하는 일은 본질적으로 부수 효과입니다. 요령은 순수한 계산 부분과 부수 효과 부분을 분리하는 것입니다. React가 이 원칙 위에 세워져 있어서(57장), 지금 익혀 두면 나중이 훨씬 수월합니다.

🛠️ 현장 노트 — 함수 작성 원칙 5가지

  1. 하나의 함수는 하나의 일만. 이름에 "and"가 들어가면 쪼갤 신호입니다.
  2. 이름은 동사로. getUserName, calculateTotal, isValid.
  3. 인자는 3개 이하. 넘으면 객체로 받습니다.
  4. 이른 반환으로 중첩을 줄입니다.
  5. 인자로 받은 객체·배열을 직접 수정하지 마세요. 호출한 쪽이 예상하지 못합니다.

📌 요약

  • 함수는 이름 붙인 레시피·자판기다. 매개변수로 받고 return으로 돌려준다. 정의만으로는 실행되지 않고 호출해야 실행된다.
  • 함수 선언문은 통째로 호이스팅되어 위에서 호출할 수 있다. 표현식·화살표는 아니다.
  • 화살표 함수는 this·arguments가 없고 new로 호출할 수 없다.
  • 객체 메서드에는 메서드 축약 문법을 쓴다. 화살표 함수는 부적절하다.
  • 화살표 함수로 객체를 반환할 때는 ({ … })로 감싼다.
  • 기본값·나머지 매개변수·객체 구조 분해로 인자를 유연하게 받는다.
  • 함수는 값이다. 변수에 담고, 넘기고, 반환할 수 있다.
  • fn은 함수 자체, fn()은 실행 결과. 콜백에 괄호를 붙이지 않는다.
  • 가능하면 순수 함수로. 부수 효과는 분리한다.
